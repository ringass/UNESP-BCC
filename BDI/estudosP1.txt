CREATE TABLE clientes(
id SERIAL PRIMARY KEY,
nome VARCHAR(32) NOT NULL,
email VARCHAR(50) NOT NULL,
idade SMALLINT
);

CREATE TABLE produtos (
id SERIAL PRIMARY KEY,
nome VARCHAR(30) NOT NULL,
preco NUMERIC(10.2) NOT NULL
);

CREATE TABLE pedidos (
id SERIAL PRIMARY KEY,
cliente_id INTEGER REFERENCES clientes(id),
produto_id INTEGER KEY REFERENCES produtos(id),
quantidade INTEGER,
data_pedido DATE
);

3 - SELECT c.nome AS cliente, p.nome AS produto, pe.quantidade
FROM pedidos pe
JOIN clientes c ON pe.cliente_id = c.id
JOIN pedidos p ON pe.produto_id = p.id;

4 - SELECT nome, preco FROM produtos WHERE (SELECT AVG(preco) FROM produtos);

5 - SELECT nome, preco FROM produtos WHERE (SELECT MAX(preco) FROM produtos);

6 - SELECT nome, preco
FROM produtos WHERE (SELECT MIN(preco) FROM produtos);

7 - SELECT c.nome AS cliente, COUNT(p.quantidade) AS total_pedidos FROM clientes c 
LEFT JOIN pedidos pe ON c.id = pe.cliente_id
GROUP BY c.nome
ORDER BY total_pedidos DESC;

8 - SELECT c.nome AS client, SUM(p.preco * pe.quantidade) AS total_gasto
FROM clientes c
JOIN pedidos pe ON pe.cliente_id = c.id
JOIN produtos p ON pe.produto_id = p.id
GROUP BY c.nome
ORDER BY total_gasto DESC;

9 - SELECT c.nome AS cliente, COUNT(pe.id) AS total_pedidos
FROM clientes c
JOIN pedidos pe ON pe.cliente_id = c.id
GROUP BY c.nome
ORDER BY total DESC
LIMIT 1;

10 - SELECT c.nome AS client, SUM(p.preco * pe.quantidade) AS total_gasto
FROM clientes c
JOIN pedidos pe ON pe.cliente_id = c.id
JOIN produtos p ON pe.produto_id = p.id
GROUP BY c.nome
ORDER BY total_gasto DESC
LIMIT 1;

11 - SELECT p.nome AS produto, COUNT(pe.id) AS produtos
FROM produtos p
LEFT JOIN pedidos pe ON pe.produtos_id = p.id
GROUP BY p.nome
ORDER BY produtos DESC
LIMIT 1;
